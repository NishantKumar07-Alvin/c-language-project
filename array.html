Array Programs in C

Here are some basic C programs based on arrays.

List of Programs

  1. Enter an array from user and display all the values
  2. Print the array in reverse order
  3. Copy the array into second array
  4. Copy array into another array in reverse order
  5. Print even numbers of array
  6. Print positive numbers of array
  7. Square all the values of array
  8. Print odd numbers from five values
  9. Add all five values of array
  10. Search a value in array
  11. Sort five values in ascending order
  12. Find maximum value of array
  13. Find minimum value of array

1. Enter an array from user and display all the values

#include <stdio.h>

int main()
{
    int a[5], i;

    printf("Enter 5 numbers:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Array values are:\n");

    for(i = 0; i < 5; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

2. Enter an array from user and print it in reverse order

#include <stdio.h>

int main()
{
    int a[5], i;

    printf("Enter 5 numbers:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Array in reverse order:\n");

    for(i = 4; i >= 0; i--)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

10. Search a value in an array

#include <stdio.h>

int main()
{
    int a[5], i, n, found = 0;

    printf("Enter 5 numbers:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter number to search: ");
    scanf("%d", &n);

    for(i = 0; i < 5; i++)
    {
        if(a[i] == n)
        {
            found = 1;
            break;
        }
    }

    if(found == 1)
    {
        printf("Found");
    }
    else
    {
        printf("Not found");
    }

    return 0;
}

11. Enter five values and sort them in ascending order

#include <stdio.h>

int main()
{
    int a[5], i, j, temp;

    printf("Enter 5 numbers:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 0; i < 5; i++)
    {
        for(j = i + 1; j < 5; j++)
        {
            if(a[i] > a[j])
            {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }

    printf("Ascending order:\n");

    for(i = 0; i < 5; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

12. Find the maximum value of an array

#include <stdio.h>

int main()
{
    int a[5], i, max;

    printf("Enter 5 numbers:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    max = a[0];

    for(i = 1; i < 5; i++)
    {
        if(a[i] > max)
        {
            max = a[i];
        }
    }

    printf("Maximum value = %d", max);

    return 0;
}
