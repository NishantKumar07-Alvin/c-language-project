1. Arithmetic Operations

Perform the given arithmetic operations one by one.

#include <stdio.h>

void add()
{
    int a, b;
    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);
    printf("Sum = %d\n", a + b);
}

void subtract()
{
    int a, b;
    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);
    printf("Difference = %d\n", a - b);
}

void multiply()
{
    int a, b;
    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);
    printf("Product = %d\n", a * b);
}

void divide()
{
    int a, b;
    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);

    if(b != 0)
        printf("Division = %.2f\n", (float)a / b);
    else
        printf("Cannot divide by zero.\n");
}

int main()
{
    add();
    subtract();
    add();
    multiply();
    subtract();
    multiply();
    divide();
    divide();
    add();

    return 0;
}

2. Sum of 10 Numbers

Enter 10 numbers and display their sum.

#include <stdio.h>

void sum()
{
    int i, num, total = 0;

    for(i = 1; i <= 10; i++)
    {
        printf("Enter number %d: ", i);
        scanf("%d", &num);
        total = total + num;
    }

    printf("Sum = %d\n", total);
}

int main()
{
    sum();
    return 0;
}

3. Return an Array from Function

In C, a simple way to share an array is to pass it to a function.

#include <stdio.h>

void display(int a[])
{
    int i;

    for(i = 0; i < 5; i++)
    {
        printf("%d ", a[i]);
    }
}

int main()
{
    int a[5] = {10, 20, 30, 40, 50};

    display(a);

    return 0;
}

4. Local and Global Variables

Examples of factorial, reverse, palindrome, prime, Armstrong and Fibonacci using a global variable.

#include <stdio.h>

int n;

void factorial()
{
    int i, fact = 1;

    for(i = 1; i <= n; i++)
        fact = fact * i;

    printf("Factorial = %d\n", fact);
}

void reverse()
{
    int num = n, digit, rev = 0;

    while(num > 0)
    {
        digit = num % 10;
        rev = rev * 10 + digit;
        num = num / 10;
    }

    printf("Reverse = %d\n", rev);
}

void palindrome()
{
    int num = n, digit, rev = 0;

    while(num > 0)
    {
        digit = num % 10;
        rev = rev * 10 + digit;
        num = num / 10;
    }

    if(rev == n)
        printf("Palindrome number\n");
    else
        printf("Not a palindrome number\n");
}

void prime()
{
    int i, flag = 0;

    if(n < 2)
        flag = 1;

    for(i = 2; i < n; i++)
    {
        if(n % i == 0)
        {
            flag = 1;
            break;
        }
    }

    if(flag == 0)
        printf("Prime number\n");
    else
        printf("Not a prime number\n");
}

void armstrong()
{
    int num = n, digit, sum = 0;

    while(num > 0)
    {
        digit = num % 10;
        sum = sum + digit * digit * digit;
        num = num / 10;
    }

    if(sum == n)
        printf("Armstrong number\n");
    else
        printf("Not an Armstrong number\n");
}

void fibonacci()
{
    int a = 0, b = 1, c, i;

    printf("Fibonacci series: ");

    for(i = 1; i <= n; i++)
    {
        printf("%d ", a);
        c = a + b;
        a = b;
        b = c;
    }

    printf("\n");
}

int main()
{
    printf("Enter a number: ");
    scanf("%d", &n);

    factorial();
    reverse();
    palindrome();
    prime();
    armstrong();
    fibonacci();

    return 0;
}

5. Swap Using Call by Address

Use pointers to swap two numbers.

#include <stdio.h>

void swap(int *a, int *b)
{
    int temp;

    temp = *a;
    *a = *b;
    *b = temp;
}

int main()
{
    int a, b;

    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);

    swap(&a, &b);

    printf("After swapping: %d %d\n", a, b);

    return 0;
}

7. Power of a Number

Enter a number and its power.

#include <stdio.h>

void power()
{
    int n, p, i;
    int result = 1;

    printf("Enter number and power: ");
    scanf("%d%d", &n, &p);

    for(i = 1; i <= p; i++)
        result = result * n;

    printf("Answer = %d\n", result);
}

int main()
{
    power();
    return 0;
}
