1. Enter a string and convert it into uppercase



#include 

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] >= 'a' && str[i] <= 'z')
        {
            str[i] = str[i] - 32;
        }
    }

    printf("Uppercase string: %s", str);

    return 0;
}

}

3. Enter a string and convert it into sentence case

#include <stdio.h>
#include <ctype.h>

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        str[i] = tolower((unsigned char)str[i]);
    }

    if(str[0] != '\0')
    {
        str[0] = toupper((unsigned char)str[0]);
    }

    printf("Sentence case: %s", str);

    return 0;
}

4. Enter a string and convert it into title case

#include <stdio.h>
#include <ctype.h>

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(i == 0 || str[i - 1] == ' ')
        {
            str[i] = toupper((unsigned char)str[i]);
        }
        else
        {
            str[i] = tolower((unsigned char)str[i]);
        }
    }

    printf("Title case: %s", str);

    return 0;
}

5. Enter a string and convert it into toggle case

#include <stdio.h>
#include <ctype.h>

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(islower((unsigned char)str[i]))
        {
            str[i] = toupper((unsigned char)str[i]);
        }
        else if(isupper((unsigned char)str[i]))
        {
            str[i] = tolower((unsigned char)str[i]);
        }
    }

    printf("Toggle case: %s", str);

    return 0;
}

12. Enter a string and copy it in reverse order

    for(i = length - 1, j = 0; i >= 0; i--, j++)

13. Enter two strings and find the difference between each character

#include <stdio.h>
#include <string.h>

{
    char str1[100], str2[100];
    int i, length;

    printf("Enter first string: ");
    fgets(str1, sizeof(str1), stdin);

    printf("Enter second string: ");
    fgets(str2, sizeof(str2), stdin);

    str1[strcspn(str1, "\n")] = '\0';
    str2[strcspn(str2, "\n")] = '\0';

    length = strlen(str1);

    if(strlen(str1) != strlen(str2))
    {
        printf("Strings must have equal length");
    }
    else
    {
        for(i = 0; i < length; i++)
        {
            printf("Difference at position %d = %d\n",
                   i, str1[i] - str2[i]);
        }
    }

    return 0;
}

14. Enter a string and check whether a character exists

        }

    else
    {
        printf("Character does not exist");
    }

}

16. Enter a string and count all vowels

#include <stdio.h>
#include <ctype.h>

int main()
{
    char str[100];
    int i, count = 0;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        char ch = tolower((unsigned char)str[i]);

        if(ch == 'a' || ch == 'e' || ch == 'i' ||
           ch == 'o' || ch == 'u')
        {
            count++;
        }
    }

    printf("Total vowels = %d", count);

    return 0;
}

18. String palindrome example using reverse

#include <stdio.h>
#include <string.h>

int main()
{
    char str[100], rev[100];
    int i, j, length;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    str[strcspn(str, "\n")] = '\0';
    length = strlen(str);

    for(i = length - 1, j = 0; i >= 0; i--, j++)
    {
        rev[j] = str[i];
    }

    rev[j] = '\0';

    printf("Reverse string: %s\n", rev);

    if(strcmp(str, rev) == 0)
    {
        printf("String is palindrome");
    }
    else
    {
        printf("String is not palindrome");
    }

    return 0;
}

19. Enter roll number, name, class and marks of 5 subjects

#include <stdio.h>
#include <string.h>

struct Student
{
    int roll;
    char name[50];
    char className[20];
    float marks[5];
};

int main()
{
    struct Student s;
    int i;

    printf("Enter roll number: ");
    scanf("%d", &s.roll);
    getchar();

    printf("Enter name: ");
    fgets(s.name, sizeof(s.name), stdin);
    s.name[strcspn(s.name, "\n")] = '\0';

    printf("Enter class: ");
    fgets(s.className, sizeof(s.className), stdin);
    s.className[strcspn(s.className, "\n")] = '\0';

    printf("Enter marks of 5 subjects:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%f", &s.marks[i]);
    }

    printf("\nStudent Details\n");
    printf("Roll Number: %d\n", s.roll);
    printf("Name: %s\n", s.name);
    printf("Class: %s\n", s.className);

    for(i = 0; i < 5; i++)
    {
        printf("Subject %d Marks: %.2f\n", i + 1, s.marks[i]);
    }

    return 0;
}
