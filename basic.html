Program 02 - Read a Character

Write a C program to read and display one character.

#include <stdio.h>

int main()
{
    char ch;

    printf("Enter a character: ");
    scanf("%c", &ch);

    printf("Character = %c", ch);

    return 0;
}

Program 03 - Read 4 Characters and Display Each on a Separate Line

Write a C program to read four characters and print each on a new line.

#include <stdio.h>

int main()
{
    char a, b, c, d;

    printf("Enter four characters: ");
    scanf(" %c %c %c %c", &a, &b, &c, &d);

    printf("%c\n", a);
    printf("%c\n", b);
    printf("%c\n", c);
    printf("%c\n", d);

    return 0;
}

Program 09 - Display Password as Stars

Write a C program to read a password and display stars instead of the characters.

#include <stdio.h>

int main()
{
    char password[20];
    int i = 0;

    printf("Enter password: ");
    scanf("%19s", password);

    printf("Password: ");

    while (password[i] != '\0')
    {
        printf("*");
        i++;
    }

    return 0;
}

Note: This prints stars after the password is entered; it does not hide characters while typing.

Program 10 - Print ASCII Values of Two Characters

Write a C program to display the ASCII values of two characters.

#include <stdio.h>

int main()
{
    char a, b;

    printf("Enter two characters: ");
    scanf(" %c %c", &a, &b);

    printf("ASCII value of %c = %d\n", a, a);
    printf("ASCII value of %c = %d\n", b, b);

    return 0;
}

Program 11 - Print ASCII Characters from 0 to 127

Write a C program to print character codes from 0 to 127.

#include <stdio.h>

int main()
{
    int i;

    for (i = 0; i <= 127; i++)
    {
        printf("%d = %c\n", i, i);
    }

    return 0;
}
